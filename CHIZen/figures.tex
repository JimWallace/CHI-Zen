%\begin{tikzpicture}
%\begin{axis}[<STYLE>,
%	% X Axis options (xlabel, xmin, xmax, xtick={})
%	% Y Axis options (ylabel, ymin, ymax, ytick={})
%       ]
%
%    % DATA GOES IN HERE (preferably read from data/*.csv)
%
%\end{axis}
%\end{tikzpicture}
%
%
% OR, for multiple plots use this template:
%
%\begin{tikzpicture}
%\begin{groupplot}[
%   group style={
%       group size= <COL> by <ROW>,
%       x descriptions at=edge bottom,
%       y descriptions at=edge left,
%       vertical sep=0pt,
%       horizontal sep=0pt},
%  <STYLE>,
%	% X Axis options (xmin, xmax, xtick={})
%	% Y Axis options (ymin, ymax, ytick={})
%      ]
%
%    % DATA GOES IN HERE
%
%\end{groupplot}
%\end{tikzpicture}
%
%
% The following styles are defined in this library:
%
%   chizen bar          clustered bar graph (comparing conditions)
%   chizen stacked bar  stacked bar graph
%   chizen box          box plot, filled boxes with a white median (distributions)
%   chizen tufte box    Tufte's minimal box plot (a dot for the median, no box)
%   chizen tufte panel  bare-bones bar panel for Tufte-style stacked panels (groupplot)
%   chizen line         line graph (trends over time)
%   chizen Likert       diverging stacked bar for Likert responses
%   range frame         Tufte's range frame (axis lines span the data only)
% and the cycle lists (palettes) chizen lines, okabe-ito, waterloo-light,
% waterloo-dark, colorbrewer-RYB and colorbrewer-RYB-plain.
%
% NAMES. The styles used to be called `IDEA ...' (after the IDEA lab), e.g.
% `IDEA bar' and the `IDEA lines' cycle list. Those names still work as
% aliases (see the end of this file), so existing papers compile unchanged,
% but new figures should use the `chizen ...' names.
%

%%%%%%%%%%%%%%%%%%%%
% GREYSCALE / COLOUR-BLIND NOTES (relative luminance, WCAG 2 formula)
%   None of the categorical palettes above is greyscale-safe: some colours
%   have almost the same luminance and print as the same grey, e.g.
%     waterloo-light: ArtsLight .336 vs SchoolRedLight .343,
%                     MathLight .441 vs ScienceLight .449
%     waterloo-dark:  AHS, Arts, Environment, Science and Math are all .17-.19
%     colorbrewer-RYB: RYB1 .564 vs RYB6 .544; RYB2 (.960) is almost white.
%   The colours are kept as they are because published figures use them.
%   For colour-blind readers use okabe-ito below; for black-and-white print,
%   also vary marks, dash patterns or hatching, or label series directly.
%%%%%%%%%%%%%%%%%%%%

% LINE GRAPHS - default for chizen line
%   Each series differs in colour, dash pattern AND mark, so lines can be told
%   apart when printed in black and white. Colours are the Okabe-Ito palette
%   (colour-blind safe), ordered darkest first so thin lines stay visible;
%   OkabeItoYellow is left out because a thin yellow line on white is too

%%%%%%%%%%%%%%%%%%%%

        alias=current axis xlabel
    },
    legend pos/outer south/.style={
        /pgfplots/legend style={
            at={%
                (%
                \@ifundefined{pgf@sh@ns@current axis xlabel}%
                {xticklabel cs:0.5}%
                {current axis xlabel.south}%
                )%
            },
            anchor=north
        }
    }
}
