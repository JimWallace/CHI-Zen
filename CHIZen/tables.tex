% Katja's dashed lines in table stuff
% arydshln rewrites LaTeX's table code and breaks the table structure in a
% tagged PDF (rows get the wrong number of cells, which fails PDF/UA), see
% https://latex3.github.io/tagging-project/tagging-status/ . So it is only
% loaded when tagging is off; with tagging on, \cdashlinelr{2-4} draws a
% solid booktabs \cmidrule instead, and \hdashline/\cdashline are not
% available.
% arydshln is not on ACM's TAPS list either, so final mode always uses
% the \cmidrule.
\makeatletter
\ifchizenfinal\expandafter\@firstoftwo\else\expandafter\IfDocumentMetadataTF\fi{%
\newcommand{\cdashlinelr}[1]{\cmidrule(lr){#1}}
}{%
\usepackage{arydshln}
\def\adl@drawiv#1#2#3{%
	\hskip.5\tabcolsep
	\xleaders#3{#2.5\@tempdimb #1{1}#2.5\@tempdimb}%
	#2\z@ plus1fil minus1fil\relax
	\hskip.5\tabcolsep}
\newcommand{\cdashlinelr}[1]{%
	\noalign{\vskip\aboverulesep
		\global\let\@dashdrawstore\adl@draw
		\global\let\adl@draw\adl@drawiv}
	\cdashline{#1}
	\noalign{\global\let\adl@draw\@dashdrawstore
		\vskip\belowrulesep}}
}
\makeatother
